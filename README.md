// MY PRINCIPLES FOR WRITING GREAT CODE

/*
  1. CLARITY OVER CLEVERNESS
     Good code is read more than written
     Explicit naming, clear structure, obvious intent
*/

function calculateAllocationWithConsequences(
  availableCash: number,
  obligations: Obligation[],
  riskTolerance: "conservative" | "moderate" | "aggressive"
): AllocationResult {
  // Not: const calc = (a, o, r) => {...}
  // But: clear function name, explicit parameters
}

/*
  2. DESIGN FOR TESTABILITY
     If code is hard to test, it's poorly designed
     Dependency injection, pure functions, clear boundaries
*/

// Testable: Pure function, no side effects
const calculateBuffer = (payroll: number, variance: number) => {
  return payroll + (variance * 0.15);
};

// Hard to test: Side effects, dependencies, unclear logic
const globalBuffer = payroll * someRandomFactor;

/*
  3. OPTIMIZE FOR READABILITY
     Code is communication
     Use clear variable names, comments for "why", simple logic
*/

// Clear intent
if (isPayrollDue() && insufficientCash()) {
  raiseFinancialStressAlert();
}

// Confusing
if (d === true && c < p) { /* ... */ }

/*
  4. MEASURE EVERYTHING
     You can't improve what you don't measure
     Logging, tracing, metrics, profiling
*/

const timer = performance.now();
const result = expensiveOperation();
const elapsed = performance.now() - timer;
logger.info(`Operation took ${elapsed}ms`);

/*
  5. FAIL GRACEFULLY
     Anticipate failure modes
     Provide clear error messages, fallbacks
*/

try {
  const result = await fetchLLMResponse(query);
